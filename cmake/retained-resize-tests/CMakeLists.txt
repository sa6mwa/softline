cmake_minimum_required(VERSION 3.24)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include(../softline_workspace.cmake)
softline_check_workspace("${root}")
project(softline_retained_resize_diagnostic LANGUAGES NONE)
find_program(BASH_EXECUTABLE bash REQUIRED)
execute_process(COMMAND "${BASH_EXECUTABLE}" "${root}/scripts/require-host-bash.sh"
  COMMAND_ERROR_IS_FATAL ANY)
if(NOT EXISTS "${SOFTLINE_RESIZE_FIXTURE}")
  message(FATAL_ERROR "Build retained_resize_fixture and set SOFTLINE_RESIZE_FIXTURE to its absolute path")
endif()
softline_check_generated_path("${root}" "${SOFTLINE_RESIZE_FIXTURE}")
find_program(TMUX_EXECUTABLE tmux REQUIRED)
enable_testing()
function(resize_case name columns rows mode variant draft)
  add_test(NAME "${name}" COMMAND "${BASH_EXECUTABLE}"
    "${root}/tests/lifecycle/retained-resize.sh" "${TMUX_EXECUTABLE}"
    "${SOFTLINE_RESIZE_FIXTURE}" "${CMAKE_BINARY_DIR}" "${columns}" "${rows}"
    "${mode}" "${variant}" "${draft}")
  set_tests_properties("${name}" PROPERTIES TIMEOUT 30 LABELS "diagnostic;retained-resize")
endfunction()
resize_case(retained_partial_shrink 60 18 retained partial late)
resize_case(retained_complete_shrink 60 18 retained complete late)
resize_case(retained_partial_width 60 30 retained partial late)
resize_case(retained_partial_height 120 18 retained partial late)
resize_case(watch_partial_shrink 60 18 watch partial late)
resize_case(yield_partial_shrink 60 18 yield partial late)
resize_case(watch_partial_early 60 18 watch partial early)
resize_case(yield_complete_early 60 18 yield complete early)
resize_case(observed_partial_shrink 60 18 retained-observed partial late)
resize_case(observed_complete_shrink 60 18 retained-observed complete late)
resize_case(observed_partial_width 60 30 retained-observed partial late)
