cmake_minimum_required(VERSION 3.24)
project(softline_terminal_test_tools LANGUAGES NONE)
include(../softline_verified_archive.cmake)
softline_configure_dependency_cache()
if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "Pinned terminal test tools require a native x86_64 Linux host")
endif()
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(extract "${root}/.cache/deps-build/x86_64-linux-gnu/terminal-tests")
set(install "${root}/.cache/deps/x86_64-linux-gnu/terminal-tests/install")
set(contract "${root}/.cache/dependency-contracts/x86_64-linux-gnu/terminal-tests.txt")
file(MAKE_DIRECTORY "${root}/.cache/dependency-contracts/x86_64-linux-gnu")
file(LOCK "${contract}.lock" GUARD PROCESS TIMEOUT 600)
file(READ "${CMAKE_CURRENT_LIST_DIR}/archives.json" manifest)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/archives.json" manifest_hash)
file(SHA256 "${root}/scripts/relocate-terminal-test-tools.py" relocation_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" provision_hash)
set(manifest_hash "${manifest_hash}:${relocation_hash}:${provision_hash}")
set(stale TRUE)
if(EXISTS "${contract}" AND EXISTS "${install}/usr/bin/Xvfb")
  file(READ "${contract}" previous)
  if(previous STREQUAL manifest_hash)
    set(stale FALSE)
  endif()
endif()
if(stale)
  # An interrupted reconstruction must never leave a successful contract.
  file(REMOVE "${contract}")
  file(REMOVE_RECURSE "${extract}" "${install}")
  file(MAKE_DIRECTORY "${extract}" "${install}")
endif()
string(JSON length LENGTH "${manifest}" archives)
math(EXPR last "${length}-1")
foreach(i RANGE ${last})
  foreach(field name url sha256 archive)
    string(JSON ${field} GET "${manifest}" archives ${i} ${field})
  endforeach()
  softline_verified_archive("terminal-test-${name}" "${url}" "${sha256}" "${archive}" verified)
  if(stale)
    file(REMOVE_RECURSE "${extract}/package")
    file(ARCHIVE_EXTRACT INPUT "${verified}" DESTINATION "${extract}/package")
    file(GLOB payload "${extract}/package/data.tar.*")
    list(LENGTH payload count)
    if(NOT count EQUAL 1)
      message(FATAL_ERROR "${name}: expected one data payload in the pinned archive")
    endif()
    file(ARCHIVE_EXTRACT INPUT "${payload}" DESTINATION "${install}")
  endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(verify_options --verify)
if(stale)
  set(verify_options "")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}"
  "${root}/scripts/relocate-terminal-test-tools.py" "${install}" ${verify_options}
  RESULT_VARIABLE relocated)
if(NOT relocated EQUAL 0)
  message(FATAL_ERROR "Could not relocate or verify pinned terminal test tools")
endif()
file(MAKE_DIRECTORY "${root}/.cache/dependency-contracts/x86_64-linux-gnu")
file(WRITE "${contract}" "${manifest_hash}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/install-root.txt" "${install}\n")
