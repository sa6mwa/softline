cmake_minimum_required(VERSION 3.24)
project(softline_local_lua LANGUAGES C)
include(../softline_local_runtime.cmake)

set(lua_hash 1c4b4068d67061f2a2231ad2b5422e77acea1487ea9890f6320af614f4373dce)
include(../softline_verified_archive.cmake)
softline_configure_dependency_cache()
if(NOT DEFINED SL_TARGET_ID OR SL_TARGET_ID STREQUAL "")
  message(FATAL_ERROR "Local Lua requires the selected lifecycle target ID")
endif()
softline_verified_archive("lua-5.5.1" "https://www.lua.org/ftp/lua-5.5.1.tar.gz"
  "${lua_hash}" lua-5.5.1.tar.gz archive)
get_filename_component(softline_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(lua_cache_root "${softline_root}/.cache")
set(lua_extract "${lua_cache_root}/deps-build/${SL_TARGET_ID}/lua")
set(lua_contract "${lua_cache_root}/dependency-contracts/${SL_TARGET_ID}/lua.txt")
set(lua_contract_value "version=5.5.1\narchive_sha256=${lua_hash}\n")
set(lua_stale FALSE)
if(NOT EXISTS "${lua_contract}")
  set(lua_stale TRUE)
else()
  file(READ "${lua_contract}" lua_existing_contract)
  if(NOT lua_existing_contract STREQUAL lua_contract_value)
    set(lua_stale TRUE)
  endif()
endif()
if(NOT EXISTS "${lua_extract}/lua-5.5.1/src/lua.h")
  set(lua_stale TRUE)
endif()
if(lua_stale)
  file(REMOVE_RECURSE "${lua_extract}")
  file(MAKE_DIRECTORY "${lua_extract}")
  file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${lua_extract}")
  get_filename_component(lua_contract_dir "${lua_contract}" DIRECTORY)
  file(MAKE_DIRECTORY "${lua_contract_dir}")
  file(WRITE "${lua_contract}" "${lua_contract_value}")
endif()
set(lua_source "${lua_extract}/lua-5.5.1/src")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lua-source-dir.txt" CONTENT "${lua_source}\n")
file(GLOB lua_sources "${lua_source}/*.c")
list(REMOVE_ITEM lua_sources "${lua_source}/lua.c" "${lua_source}/luac.c")
add_library(local_lua STATIC ${lua_sources})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(local_lua PUBLIC LUA_USE_LINUX)
elseif(APPLE)
  target_compile_definitions(local_lua PUBLIC LUA_USE_MACOSX)
elseif(UNIX)
  target_compile_definitions(local_lua PUBLIC LUA_USE_POSIX)
endif()
target_link_libraries(local_lua PUBLIC ${CMAKE_DL_LIBS})
if(UNIX AND NOT APPLE)
  target_link_libraries(local_lua PUBLIC m)
endif()
add_executable(lua "${lua_source}/lua.c")
target_link_libraries(lua PRIVATE local_lua)
if(APPLE)
  # LUA_USE_MACOSX enables the upstream interpreter's readline integration.
  target_link_libraries(lua PRIVATE readline)
endif()
set_target_properties(lua PROPERTIES ENABLE_EXPORTS ON)
set_property(TARGET lua APPEND PROPERTY BUILD_RPATH "${SOFTLINE_LOCAL_LIBRARY_DIR}")
softline_local_runtime(lua)
